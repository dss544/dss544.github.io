xv6实验[3]-写时复制

2026-06-11 8 min read

1、为什么需要COW？

在xv6原生的fork()系统调用中，内核会采用“无脑硬拷贝”的策略：遍历父进程的所有虚拟内存，分配同等数量的物理页，并使用memmove将数据逐字节拷贝给子进程。
这种设计存在两个致命缺陷：
1、极高的时空开销：对于占用内存较大的进程，完整的内存拷贝会导致极大的延迟，并瞬间消耗大量物理内存。
2、严重的资源浪费：在 Unix/Linux 生态中，fork() 之后紧接 exec()（加载新程序）是最常见的进程创建范式。原先费时费力拷贝过来的物理页数据，在 exec() 执行的瞬间就会被清空作废。

为解决这一问题，程序员们引入了写时复制（Copy-on-Write, COW）机制：在 fork() 时只拷贝页表而不分配物理页帧；父子进程共享同一块物理内存，并将其页表项（PTE）标记为只读。只有当某一方尝试写入数据触发缺页异常时，内核才进行真正的内存分配与拷贝。

2、核心架构重构

2.1 改造 fork 与 uvmcopy：共享与降权

COW机制的第一步是修改fork的底层复制逻辑。我们重写uvmcopy函数，使其不再调用kalloc 分配新内存，而是直接将子进程的虚拟地址映射到父进程当前的物理页上。
关键在于权限管理：我们需要剥夺父子进程的写权限（清除 PTE_W），并打上自定义的软件标记 PTE_COW，以便后续在异常处理时识别。同时，由于物理页被多个进程共享，必须增加其引用计数。

int
uvmcopy(pagetable_t old, pagetable_t new, uint64 sz)
{
  pte_t *pte;
  uint64 pa, i;
  uint flags;

  for(i = 0; i < sz; i += PGSIZE){
    if((pte = walk(old, i, 0)) == 0)
      panic("uvmcopy: pte should exist");
    if((*pte & PTE_V) == 0)
      panic("uvmcopy: page not present");
    pa = PTE2PA(*pte);
    flags = PTE_FLAGS(*pte);
    // 只有拥有可写权限才赋予COW标志
    if (flags & PTE_W)
    {
      flags = (flags & ~PTE_W) | PTE_COW;
      *pte = PA2PTE(pa) | flags;
    }
    // 子进程页表映射父亲当前页映射的物理地址
    if(mappages(new, i, PGSIZE, (uint64)pa, flags) != 0){
      goto err;
    }
    // 添加引用
    kaddref((void *)pa);
  }
  sfence_vma();
  return 0;
 err:
  uvmunmap(new, 0, i / PGSIZE, 1);
  return -1;
}

2.2 物理页生命周期管理：引用计数机制

为了安全回收内存，必须引入引用计数体系。实际可用的物理内存是从KERNBASE到PHYSTOP（共计 128MB）。我们可以通过偏移量映射出一个静态数组 page_ref，记录每个物理页帧的被映射次数。

#define PA2INDEX(pa) (((uint64)pa - KERNBASE) / PGSIZE)
int page_ref[(PHYSTOP - KERNBASE) / PGSIZE];

// 增加引用
int kaddref(void *pa) {
  int ref = 0;
  acquire(&ref_lock);
  ref = ++page_ref[PA2INDEX(pa)];
  release(&ref_lock);
  return ref;
}

// 减少引用
int ksubref(void *pa) {
  int ref = 0;
  acquire(&ref_lock);
  if (page_ref[PA2INDEX(pa)] > 0) {
    ref = --page_ref[PA2INDEX(pa)];
  }
  release(&ref_lock);
  return ref;
}

配套修改kfree函数，确立“递减-判断-回收”的逻辑：只有当引用计数降至 0 时，才允许使用垃圾数据填充并挂入空闲链表。。

void kfree(void *pa) {
  struct run *r;
  if(((uint64)pa % PGSIZE) != 0 || (char*)pa < end || (uint64)pa >= PHYSTOP)
    panic("kfree");
  // 第一步：安全递减引用计数
  int ref = ksubref(pa);
  if (ref > 0) {
      return; // 仍有进程共享，直接返回
  }
  // 第二步：彻底回收并放回 freelist
  memset(pa, 1, PGSIZE); 
  r = (struct run*)pa;
  acquire(&kmem.lock);
  r->next = kmem.freelist;
  kmem.freelist = r;
  release(&kmem.lock);
}

3. 缺页拦截与按需分配 (Demand Allocation)

3.1 捕获 Store Page Fault

当进程企图写入一个被标记为只读的 COW 页时，RISC-V 的 MMU 会触发写缺页异常（scause == 15）。在 kernel/trap.c 的 usertrap 函数中，我们需要对此异常进行精准拦截与分发。
架构细节： 注意处理系统调用（scause == 8）时，我们将 PC 指针推到了下一条指令（epc += 4）。但在处理缺页异常时不需要这么做，因为分配完物理页并恢复写权限后，我们需要让 CPU 重新执行那条曾经失败的写入指令。

void usertrap(void) {
  // ... 前置鉴权与上下文保存 ...
  if(r_scause() == 8){
    // 处理系统调用
    p->trapframe->epc += 4;
    intr_on();
    syscall();
  }
  // 捕获写缺页异常，且该页面必须是合法的 COW 页面
  else if(r_scause() == 15 && uvmcheckcowpage(p->pagetable, r_stval())) {
      if (cow_alloc(p->pagetable, r_stval()) == -1) {
        p->killed = 1; // 内存分配失败（OOM），终止进程
      }
  }
  // ...
}

这里封装了一个函数uvmcheckcowpage，确保触发异常的虚拟地址位于合法范围内，并且确实携带有 PTE_COW 标记。

int uvmcheckcowpage(pagetable_t pagetable, uint64 va)
{
  pte_t *pte;
  return va < MAXVA && ((pte = walk(pagetable, va, 0)) != 0)
  && (*pte & PTE_V)
  && (*pte & PTE_U)
  && (*pte & PTE_COW);
}

3.2 极致的无拷贝优化：cow_alloc 与 kcopy_n_deref

这是整个 COW 机制中最精妙的部分。当异常发生时，如果该物理页的引用计数已经降为 1（说明其他共享进程均已退出，当前进程处于独占状态），我们完全不需要申请新内存，直接在原物理页上恢复 PTE_W 权限即可。如果不是1，就老老实实从内存里面找到找到一个未被使用的物理页，将原物理页的内容拷贝过去。

void * kcopy_n_deref(void *pa) {
  // 终极优化：独占页面免拷贝
  if (readref(pa) <= 1) {
    return pa; 
  }
  // 共享页面：分配新页并复制数据
  uint64 newpa = (uint64)kalloc();
  if (newpa == 0) return 0;
  memmove((void*)newpa, (void*)pa, PGSIZE);
  kfree(pa); // 释放原页面的引用
  return (void *)newpa;
}

int cow_alloc(pagetable_t pagetable, uint64 va) {
  pte_t *pte;
  if ((pte = walk(pagetable, va, 0)) == 0) panic("cow_alloc: walk");
    
  uint64 pa = PTE2PA(*pte);
  uint64 newpa = (uint64)kcopy_n_deref((void*)pa); 
  if (newpa == 0) return -1; 
  // 直接原地修改 PTE，极致高效，且无视 va 不对齐问题
  uint64 flags = (PTE_FLAGS(*pte) | PTE_W) & ~PTE_COW;
  *pte = PA2PTE(newpa) | flags;
  sfence_vma(); // 必须刷新 TLB
  return 0;
}

3.3 补齐内核态盲区：copyout 防护

系统的缺页异常机制通常只能捕获用户态的非法写入。如果内核代码（如通过 read 系统调用读取文件时）使用 copyout 函数，它会通过软件寻址直接绕过 CPU 的内存保护机制，从而污染共享的物理页。
为此，必须在 copyout 中前置鉴权，强制触发 COW 内存分配：

int copyout(pagetable_t pagetable, uint64 dstva, char *src, uint64 len) {
  uint64 n, va0, pa0;
  while(len > 0){
    va0 = PGROUNDDOWN(dstva);
    // 前置拦截：如果是 COW 页，提前触发复制与权限变更
    if (uvmcheckcowpage(pagetable, va0)) {
      if (cow_alloc(pagetable, va0) == -1) return -1;
    }
    pa0 = walkaddr(pagetable, va0);
    if(pa0 == 0) return -1;
    n = PGSIZE - (dstva - va0);
    if(n > len) n = len;
    memmove((void *)(pa0 + (dstva - va0)), src, n);
    len -= n;
    src += n;
    dstva = va0 + PGSIZE;
  }
  return 0;
}

从底层实现的角度来看，COW 机制不仅仅是一种节约内存的策略，更体现了现代工业级操作系统设计中“懒惰（Lazy）即高效”的核心哲学。这种将昂贵的资源分配操作尽可能推迟（甚至避免）的思想，正是构建高性能底层系统的基石。最后也是通过了实验，这个实验花了一天时间，继续努力吧，下一个实验是多线程。
