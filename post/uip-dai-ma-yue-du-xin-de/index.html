PT_THREAD(my_task(struct pt *pt)) {
    PT_BEGIN(pt);   // 展开就是：LC_RESUME(pt->lc); 即 switch(pt->lc) { case 0:

    step1_do_something();

    PT_WAIT_UNTIL(pt, condition_ok()); // 展开包含：LC_SET(pt->lc); 

    step2_do_something();

    PT_END(pt);     // 展开就是：LC_END(pt->lc); 即 }
}

定点数思想应用：

      if(uip_connr->nrtx == 0) {  
        signed char m;
        m = uip_connr->rto - uip_connr->timer;
        /* This is taken directly from VJs original code in his paper */
        m = m - (uip_connr->sa >> 3);
        /* 估计平均RTT  sa=8*RTT*/
        /* 这里其实就是RTT_new = 7/8*RTT_old + 1/8m(RTT) 
          * 典型的定点数（Fixed-point）思想 
          * sa 放大 8 倍，相当于保留 3 位二进制小数精度
        */
        uip_connr->sa += m;
        if(m < 0) {
          m = -m;
        }
        m = m - (uip_connr->sv >> 2);
        /* 波动  sv = 4*RTT偏差*/
        uip_connr->sv += m;
        uip_connr->rto = (uip_connr->sa >> 3) + uip_connr->sv;
