  1. References（库引用）
    对应概念： 相当于 C/C++ 里的 #include 头文件与链接库（.lib / .so）。

  2. DUTs（Data Unit Types，自定义数据类型）
    对应概念： 相当于 C/C++ 里的 typedef struct、enum、union。

  3. GVLs（Global Variable Lists，全局变量表）
    对应概念： 相当于 C/C++ 里的全局变量声明文件（extern int g_val;）。

  4. POUs（Program Organization Units，程序组织单元）
    对应概念： 相当于你的源代码文件夹（.cpp / .c 源文件）。

  5. VISUs（Visualizations，可视化界面）
    对应概念： 相当于上位机轻量级 UI / HMI 画面

  6. External Types（外部类型）
    作用： 当你引入了 C++ 模块、MATLAB/Simulink 生成的模型或外部 TMC 接口描述文件时，外部模块导出的数据类型会显示在这里，供 PLC 侧调用。

  7. PlcTask (PlcTask)（PLC 实时任务配置）
    对应概念： 相当于 RTOS 中的线程/任务调度配置器。

  8. PLC_Project1 Instance（PLC 运行实例）
    作用：上面的 PLC_Project1 Project 是你的“源码工程”，而 Instance 是该工程在实时内核中的“运行实例镜像”。

源性plc相当于配置下拉（正常0V），外部需要配置输出高电平传感器（PNP输出24V），漏极相当于配置上拉（正常24V），外部需要配置低电平传感器（NPN输出0V）。

Q（Switching device，开闭/分断器件）：用于通断主电路或大功率电路。

QS：隔离开关 / 刀开关（S 代表 Disconnector/Switch，主要用于检修时形成可见断点、无灭弧或弱灭弧能力）。

QF：断路器 / 空开（F 代表 Protection/Fault，具备短路和过载快速切断保护能力的开关）。

S（Switching device for control circuit，控制回路开关）：用于传输控制信号，不承载动力大电流。

SB：操作按钮（B 代表 Button，如急停按钮、启动/停止按钮）。

SQ：行程开关 / 限位开关（位置检测）。

K（Relay / Contactor，继电器与接触器）：

KM：交流接触器（M 代表 Motor，用于通断电机等大电流设备主电源）。

KA：中间继电器（A 代表 Auxiliary 辅助，用于单片机/PLC 弱电与强电之间的电气隔离与信号中继）。

KT：时间继电器。

T & U（电能转换）：

TC / T：控制变压器（如 380V 转 220V）。

U / VC：开关电源（AC 220V 转 DC 24V 整流稳压模块）。

F（Protection device，保护器件）：

FU：熔断器 / 保险丝。

FR：热过载继电器（用于电机过载发热保护）。

X & W（连接与线缆）：

XT / X：接线端子排。

W：导线、电缆、总线。

看了光电所plc的错误处理逻辑经常不好懂 而且写很多 自己想了一个错误逻辑或许可以简化点

流程图


┌── 轨道 A：[动作 1 (耗时)] ──> [动作 2 (依附 1)] ──┐
               │                                                  │
[触发异常 990] ─┤                                                  ├──> [汇合完成 999]
 (双轨并发启动) │                                                  │
               └── 轨道 B：[动作 3] ───────> [动作 4] ──> [动作 5] ┘

示例代码

VAR
    sub_step_A : INT; // 轨道 A 步号 (动作 1 -> 动作 2)
    sub_step_B : INT; // 轨道 B 步号 (动作 3 -> 动作 4 -> 动作 5)
    bTrackA_Done : BOOL;
    bTrackB_Done : BOOL;
    step_timeout_ton : TON;
END_VAR

CASE expose_scan_i OF
    (* ================= 990: 触发双轨并行异常收尾 ================= *)
    990:
        sub_step_A   := 1;     // 激活轨道 A
        sub_step_B   := 1;     // 激活轨道 B
        bTrackA_Done := FALSE;
        bTrackB_Done := FALSE;
        
        step_timeout_ton(IN := TRUE, PT := T#5S); // 总看门狗防死锁
        expose_scan_i := 991;  // 转入双轨执行与汇合等待

    (* ================= 991: 双轨并行推进与最终 Join ================= *)
    991:
        step_timeout_ton();

        (* ------------- 轨道 A 逻辑 (1 耗时 -> 2 依赖 1) ------------- *)
        IF NOT bTrackA_Done THEN
            CASE sub_step_A OF
                1: (* 动作 1：耗时动作 (例如轴减速停止) *)
                    fb_Stop[Axis_E](Execute := TRUE, Deceleration := 2000.0);
                    IF fb_Stop[Axis_E].Done OR fb_Stop[Axis_E].Error THEN
                        fb_Stop[Axis_E](Execute := FALSE);
                        sub_step_A := 2; // 动作 1 彻底完成后，才准进动作 2
                    END_IF;

                2: (* 动作 2：强依赖动作 1 (例如轴停稳后，抱闸锁死/关伺服使能) *)
                    Axis_Struct[Axis_E].i_xEnable := FALSE; // 关使能
                    bTrackA_Done := TRUE;                   // 轨道 A 全部收尾完成
            END_CASE;
        END_IF;

        (* ------------- 轨道 B 逻辑 (3 -> 4 -> 5 顺序流转) ------------- *)
        IF NOT bTrackB_Done THEN
            CASE sub_step_B OF
                1: (* 动作 3：例如立即关快门/切断紫外光源 *)
                    DO_Shutter := FALSE;
                    sub_step_B := 2; // 立即转入动作 4，根本不等轨道 A 的电机

                2: (* 动作 4：例如关辅助吹气/关气阀 *)
                    DO_BlowGas := FALSE;
                    sub_step_B := 3; // 转入动作 5

                3: (* 动作 5：纯软件标志与计数器复位 *)
                    expose_auto_cnt   := 0;
                    expose_auto_cnt_i := 0;
                    expose_auto       := 0;
                    expose_auto_flg   := nErrorCode;
                    bTrackB_Done      := TRUE; // 轨道 B 全部收尾完成
            END_CASE;
        END_IF;

        (* ------------- 双轨汇合点 (Join) ------------- *)
        IF (bTrackA_Done AND bTrackB_Done) OR step_timeout_ton.Q THEN
            step_timeout_ton(IN := FALSE);
            expose_scan_i := 999; // 所有硬件与软件全部落入安全态，停靠报警
        END_IF;

    (* ================= 999: 报警驻留步 ================= *)
    999:
        IF bAlarmReset THEN
            nErrorCode    := 0;
            expose_scan_i := 0;
        END_IF;
END_CASE;
