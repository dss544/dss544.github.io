二分查找升级 减少cache和判断 找到第一个大于等于和最后一个大于等于

class BinarySearchTemplates {
public:
    // 1. 查找【第一个等于 t】的下标 (若不存在返回 -1)
    static int findFirstEqual(const int x[], int n, int t) {
        int l = -1, u = n;
        while (l + 1 != u) {
            int m = l + (u - l) / 2;
            if (x[m] >= t) u = m;
            else           l = m;
        }
        if (u < n && x[u] == t) return u;
        return -1;
    }

    // 2. 查找【第一个大于 t】的下标 (Upper Bound，不存在返回 -1)
    static int findFirstGreater(const int x[], int n, int t) {
        int l = -1, u = n;
        while (l + 1 != u) {
            int m = l + (u - l) / 2;
            if (x[m] > t) u = m;
            else          l = m;
        }
        if (u < n) return u; // u 就是第一个大于 t 的位置
        return -1;
    }

    // 3. 查找【最后一个等于 t】的下标 (即你写的代码，若不存在返回 -1)
    static int findLastEqual(const int x[], int n, int t) {
        int l = -1, u = n;
        while (l + 1 != u) {
            int m = l + (u - l) / 2;
            if (x[m] > t) u = m;
            else          l = m;
        }
        if (l > -1 && x[l] == t) return l;
        return -1;
    }

    // 4. 查找【最后一个小于等于 t】的下标 (不存在返回 -1)
    static int findLastLessOrEqual(const int x[], int n, int t) {
        int l = -1, u = n;
        while (l + 1 != u) {
            int m = l + (u - l) / 2;
            if (x[m] > t) u = m;
            else          l = m;
        }
        if (l > -1) return l; // l 就是最后一个小于等于 t 的位置
        return -1;
    }
};

// =========================================================
// 版本 1：固定步长折半循环版本 (while 循环)
// =========================================================
int binary_search_1000_loop(const int x[], int t) {
    int i = 512;
    int l = -1;
    // 1. 首步预处理对齐：确保后续步长 i 严格为 2 的幂 (512)
    if (x[511] < t) {
        l = 1000 - 512; // l = 488，此时右边界 l + i = 1000
    }
    // 2. 步长依次折半：512 -> 256 -> 128 -> ... -> 1
    while (i != 1) {
        int nexti = i >> 1; // 相当于 nexti = i / 2，使用位移指令更快
        if (x[l + nexti] < t) {
            l += nexti;
        }
        i = nexti;
    }
    // 3. 循环结束时 i == 1，l 为最后一个小于 t 的下标
    // p = l + 1 即为第一个大于等于 t 的候选位置
    int p = l + 1;
    if (p >= 1000 || x[p] != t) {
        return -1; // 未找到
    }
    return p; // 找到目标，返回第一次出现的下标
}

// =========================================================
// 版本 2：终极优化 —— 9 次完全循环展开 (无循环，绝对无跳转开销)
// =========================================================
int binary_search_1000_unrolled(const int x[], int t) {
    int l = -1;
    // 1. 首步预处理对齐
    if (x[511] < t) {
        l = 1000 - 512; // 488
    }
    // 2. 彻底展开 9 次判断，无需 while 循环判断和 i 计数器
    // 跨度：256 -> 128 -> 64 -> 32 -> 16 -> 8 -> 4 -> 2 -> 1
    if (x[l + 256] < t) l += 256;
    if (x[l + 128] < t) l += 128;
    if (x[l + 64]  < t) l += 64;
    if (x[l + 32]  < t) l += 32;
    if (x[l + 16]  < t) l += 16;
    if (x[l + 8]   < t) l += 8;
    if (x[l + 4]   < t) l += 4;
    if (x[l + 2]   < t) l += 2;
    if (x[l + 1]   < t) l += 1;
    // 3. 检查最终匹配结果
    int p = l + 1;
    if (p >= 1000 || x[p] != t) {
        return -1;
    }
    return p;
}

// 利用无符号数特性：小于 '0' 转成大数，从而只用一次比较
bool is_digit_fast(char c) {
    return (unsigned int)(c - '0') < 10;
}

bool is_upper_fast(char c) {
    return (unsigned int)(c - 'A') < 26;
}

bool is_lower_fast(char c) {
    return (unsigned int)(c - 'a') < 26;
}
