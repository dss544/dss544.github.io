xv6实验[7]-locks

2026-07-06 10 min read

Memory allocator

第一个实验是对XV6内核的内存页面分配器进行改进，改进的策略在前面的章节中也讲过了。XV6原本是使用一个空闲页面链表，但是这样就会导致不同CPU上的kalloc和kfree会产生锁争用，内存页面的分配被完全串行化了，降低了系统的性能。
而一个改进策略就是为每个CPU核心分配一个空闲链表，kalloc和kfree都在本核心的链表上进行，只有当当前核心的链表为空时才去访问其他核心的链表。通过这种策略就可以减少锁的争用，只有当某核心的链表为空时才会发生锁争用。
首先定义NCPU个kmem结构体，并在kinit函数中对锁进行初始化。

struct  {
  struct spinlock lock;
  struct run *freelist;
  char localname[10];
}  kmems[NCPU];
void
kinit()
{
  for (int i = 0; i < NCPU; i++) {
    snprintf(kmem[i].lock_name, sizeof(kmem[i].lock_name), "kmem_%d", i);
    initlock(&kmem[i].lock, kmem[i].lock_name);
  }
  freerange(end, (void*)PHYSTOP);
}

对于kfree函数，直接把块释放到该cpu的链表上。

void
kfree(void *pa)
{
 struct run *r;

 if(((uint64)pa % PGSIZE) != 0 || (char*)pa < end || (uint64)pa >= PHYSTOP)
   panic("kfree");

 // Fill with junk to catch dangling refs.
 memset(pa, 1, PGSIZE);

 r = (struct run*)pa;
 // 直接释放到各个cpu上的链表上

 push_off();
 int id = cpuid();
 acquire(&kmems[id].lock);
 r->next = kmems[id].freelist;
 kmems[id].freelist = r;
 release(&kmems[id].lock);
 pop_off();
}

对于kalloc函数，首先在自己的链表里面寻找，如果找到直接返回，如果没有遍历其他cpu，如果其他cpu有空闲就摘除他们的一半来填充自己，如果他们只有一个也能用。

void *
kalloc(void)
{
 struct run *r;
 push_off();
 int id = cpuid();
 acquire(&kmems[id].lock);
 r = kmems[id].freelist;
 // 在自己这里拿到了
 if(r){
   kmems[id].freelist = r->next;
   release(&kmems[id].lock);
 }
 else
 {
   release(&kmems[id].lock);
   // 去其他cpu找找
   for (int i = 0; i < sizeof(kmems) / sizeof(*kmems); i++)
   {
     if (i == id) continue;
     acquire(&kmems[i].lock);
     // 有东西
     if (kmems[i].freelist)
     {
       struct run **slow = &kmems[i].freelist;
       struct run *fast = kmems[i].freelist;
       while(fast && fast->next)
       {
         fast = fast->next->next;
         slow = &((*slow)->next);
       }   
         r = *slow;
         *slow = 0;
         release(&kmems[i].lock);
         acquire(&kmems[id].lock);
         kmems[id].freelist = r->next;
         release(&kmems[id].lock);
         break;
     }
     release(&kmems[i].lock);
   }
 }
 pop_off();
 if(r)
   memset((char*)r, 5, PGSIZE); // fill with junk
 return (void*)r;
}

Buffer cache

为了提高并行性能，我们可以用哈希表来代替链表，这样每次获取和释放的时候，都只需要对哈希表的一个桶进行加锁，桶之间的操作就可以并行进行。只有当需要对缓冲区进行驱逐替换时，才需要对整个哈希表加锁来查找要替换的块。
使用哈希表就不能使用链表来维护LRU信息，因此需要在buf结构体中添加timestamp域来记录释放的事件，同时prev域也不再需要。

struct buf {
    ...
    ...
 uint timestamp;
};

在brelse函数中记录timestamp

void
brelse(struct buf *b)
{
 if(!holdingsleep(&b->lock))
   panic("brelse");

 releasesleep(&b->lock);

 // 8. 【核心修复】：彻底废除全局锁，释放时只需要拿该块对应的哈希桶锁！
 int idx = hash(b->dev, b->blockno);
 acquire(&hashtable[idx].lock);
 b->refcnt--;
 if (b->refcnt == 0) {
   b->timestamp = ticks; // 盖上本地时间戳
 }
 release(&hashtable[idx].lock);
}

定义哈希表的结构体，bcache.lock为表级锁，而hashtable[i].lock为桶级锁：

#define NBUCKET 13
#define NBUF (NBUCKET * 3)

struct {
  struct spinlock lock;
  struct buf buf[NBUF];
} bcache;

struct bucket {
  struct spinlock lock;
  struct buf head;
}hashtable[NBUCKET];

int
hash(uint dev, uint blockno)
{
  return blockno % NBUCKET;
}

在binit函数中对哈希表进行初始化，将bcache.buf[NBUF]中的块平均分配给每个桶，记得设置b->blockno使块的hash与桶相对应，后面需要根据块来查找对应的桶。

#define NBUCKET 13
#define NBUF          (NBUCKET * 3)
void
binit(void)
{
  struct buf *b;

  initlock(&bcache.lock, "bcache");

  for(b = bcache.buf; b < bcache.buf+NBUF; b++){
    initsleeplock(&b->lock, "buffer");
  }

  // 1. 【修复】：让哈希桶的链表头初始化为双向自循环，一个都不能漏
  for (int i = 0; i < NBUCKET; i++) {
    initlock(&hashtable[i].lock, "bcache_bucket");
    hashtable[i].head.next = &hashtable[i].head;
    hashtable[i].head.prev = &hashtable[i].head;
  }

  // 2. 【修复】：不要用 NBUF/NBUCKET，直接遍历 所有块，取模打散
  for (int i = 0; i < NBUF; i++) {
    b = &bcache.buf[i];
    int bucket_id = i % NBUCKET;
    b->blockno = bucket_id; // 开机哈希对齐
    b->dev = 0;
    b->refcnt = 0;
    b->timestamp = 0;

    // 严谨的双向循环链表头插
    struct buf *head = &hashtable[bucket_id].head;
    b->next = head->next;
    b->prev = head;
    head->next->prev = b;
    head->next = b;
  }
}

之后就是重点bget函数，首先在对应的桶当中查找当前块是否被缓存，如果被缓存就直接返回；如果没有被缓存的话，就需要查找一个块并将其逐出替换。我这里使用的策略是先在当前桶当中查找，当前桶没有查找到再去全局数组中查找，这样的话如果当前桶中有空闲块就可以避免全局锁。
在全局数组中查找时，要先加上表级锁，当找到一个块之后，就可以根据块的信息查找到对应的桶，之后再对该桶加锁，将块从桶的链表上取下来，释放锁，最后再加到当前桶的链表上去。
这里有个小问题就是全局数组中找到一个块之后，到对该桶加上锁之间有一个窗口，可能就在这个窗口里面这个块就被那个桶对应的本地查找阶段用掉了。因此，需要在加上锁之后判断是否被用了，如果被用了就要重新查找。

static struct buf*
bget(uint dev, uint blockno)
{
  struct buf *b;
  int idx = hash(dev, blockno);
  struct bucket* bucket = hashtable + idx;
  acquire(&bucket->lock);
  // 第一阶段：本桶高速命中
  for(b = bucket->head.next; b != &bucket->head; b = b->next){
    if(b->dev == dev && b->blockno == blockno){
      b->refcnt++;
      release(&bucket->lock);
      acquiresleep(&b->lock);
      return b;
    }
  }
  // 第二阶段：本桶局部替换
  uint min_time = 0xffffffff;
  struct buf* replace_buf = 0;
  for(b = bucket->head.next; b != &bucket->head; b = b->next){
    if(b->refcnt == 0 && b->timestamp < min_time) {
      replace_buf = b;
      min_time = b->timestamp;
    }
  }
  if(replace_buf) {
    goto find; // 本桶有货，直接跳去 find
  }
  // 第三阶段：全局偷块
  // 3. 【核心修复】：去全局抢块前，先放掉当前桶的锁，消除 A-B-B-A 死锁！
  release(&bucket->lock);
  acquire(&bcache.lock);
refind:
  replace_buf = 0;
  min_time = 0xffffffff;
  for(b = bcache.buf; b < bcache.buf + NBUF; b++) {
    if(b->refcnt == 0 && b->timestamp < min_time) {
      replace_buf = b;
      min_time = b->timestamp;
    }
  }
  if (replace_buf) {
    int ridx = hash(replace_buf->dev, replace_buf->blockno);
    // 4. 【核心修复】：严格按照编号从小到大对两个桶加锁，防御并发死锁
    if (ridx < idx) {
      acquire(&hashtable[ridx].lock);
      acquire(&hashtable[idx].lock);
    } else {
      acquire(&hashtable[idx].lock);
      acquire(&hashtable[ridx].lock);
    }
    if(replace_buf->refcnt != 0) {
      release(&hashtable[ridx].lock);
      release(&hashtable[idx].lock);
      goto refind;
    }
    // 5. 【核心修复】：采用严谨的双向循环链表摘除，防止内存野指针 Panic
    replace_buf->next->prev = replace_buf->prev;
    replace_buf->prev->next = replace_buf->next;
    release(&hashtable[ridx].lock); // 摘除后立刻释放老家锁
    // 6. 【核心修复】：采用严谨的双向循环链表挂载到新家
    replace_buf->next = bucket->head.next;
    replace_buf->prev = &bucket->head;
    bucket->head.next->prev = replace_buf;
    bucket->head.next = replace_buf;
    // 7. 【核心修复】：大搬家完，在 goto find 前立刻释放全局大锁！
    // 这样在接下来的 acquiresleep 睡觉读盘期间，全局锁是放开的，多核绝不卡死。
    release(&bcache.lock);
    goto find;
  }
  else {
    release(&bcache.lock);
    panic("bget: no buffers");
  }
find:
  // 此时手里唯一抓着当前桶的锁（bucket->lock）
  replace_buf->dev = dev;
  replace_buf->blockno = blockno;
  replace_buf->valid = 0;
  replace_buf->refcnt = 1;
  release(&bucket->lock); // 安全释放桶锁
  acquiresleep(&replace_buf->lock);
  return replace_buf;
}

最后将bpin和bunpin的锁替换为桶级锁就行了：

void
bpin(struct buf *b) {
  int idx = hash(b->dev, b->blockno);
  acquire(&hashtable[idx].lock);
  b->refcnt++;
  release(&hashtable[idx].lock);
}

void
bunpin(struct buf *b) {
  int idx = hash(b->dev, b->blockno);
  acquire(&hashtable[idx].lock);
  b->refcnt--;
  release(&hashtable[idx].lock);
}
