    uint8_t diff = 0;

unsigned int add_with_carry(unsigned int a, unsigned int b, unsigned char *carry) {
    unsigned int sum = b; // 先把 b 赋给 sum，作为相加的底数
    unsigned char c;
    __asm__ (
        "addl %[in_a], %[out_sum]\n\t"  // out_sum = out_sum + in_a
        "setc %[out_c]\n\t"             // 如果产生无符号进位(CF=1)，out_c 设为 1，否则 0
        : [out_sum] "+r" (sum),         // "+" 表示该寄存器可读可写：执行前读入初始值，执行后写回 sum
          [out_c]   "=r" (c)            // "=" 表示只写输出
        : [in_a]    "r"  (a)            // "r" 表示输入变量 a
        : "cc"                          // 通知编译器修改了条件标志位
    );
    *carry = c;
    return sum;
}

读在前，写在后或者读写完全不沾边效率高，先写在读效率低下，因为接下来指令需要把这个地址新值读出来，卡住流水线。代码示例如下：

void smooth_bad(long *arr, long n) {
    for (long i = 1; i < n; i++) {
        // 第 i 轮：把新值写入 arr[i]
        arr[i] = (arr[i] + arr[i - 1]) / 2
        // 🚨 灾难发生了：
        // 当循环进入第 i + 1 轮时，CPU 必须立刻读取 arr[i]！
        // 这一轮的 Load 必须死等上一轮的 Store 转发数据，跨迭代形成死锁链。
    }
}
void smooth_good(long *arr, long n) {
    long prev = arr[0]; // 提前把旧值读入寄存器
    
    for (long i = 1; i < n; i++) {
        long curr = arr[i];          // 1. 【先读】提前把原数组的旧值捞进寄存器
        arr[i] = (curr + prev) / 2;  // 2. 【后写】写入新值到内存，单向发射进写缓冲区
        prev = curr;                 // 3. 将原旧值流转给下一轮（纯寄存器赋值，0 延迟）
    }
}


typedef struct {
    long x, y;
    long vx, vy;
} Particle;

void update_particles_bad(Particle *p, long n) {
    for (long i = 0; i < n; i++) {
        // 1. 先写内存：更新速度
        p[i].vx += 5; 
        // 2. 紧接着读内存：利用刚写进去的速度更新位置
        // 编译器生成汇编：先 Store vx 到结构体，紧接着下一条指令又 Load vx 出去算位置！
        p[i].x += p[i].vx; 
    }
}
void update_particles_good(Particle *p, long n) {
    for (long i = 0; i < n; i++) {
        // 1. 【先读】把数据一口气读入局部变量（寄存器）
        long vx = p[i].vx + 5;
        long x  = p[i].x + vx; // 直接在寄存器中接力计算，完全不经过内存
        
        // 2. 【后写】计算全部完成，一次性扔给写缓冲区
        p[i].vx = vx;
        p[i].x  = x;
    }
}

对于cacheline，假如一级只有64字节，读取到一个没访问字节就是比如65，就会把64~127全部读取进来。
伪共享 ：两个不同的 CPU 核心分别读写两个“在逻辑上毫无关系”的独立变量，但这俩变量在物理内存中恰好挨得太近，被打包塞进了“同一个 64 字节的 Cache Line”里，导致多核硬件缓存一致性协议在后台疯狂打架，性能断崖式暴跌。解决方案放在不同cacheline里面。

在替换缓存块的时候，有几种策略，比如任意排布每次遍历，或者通过hash分组。一般都是分成几组，然后在组里找，在组里实现类似LRU算法。

CPU 读取 1 个字节时的硬件三步走：（P426页）
1、组选择（Set Selection）：硬件用地址中间的 sss 位确定组号（Cache Set）。
2、 行匹配（Line Matching）：硬件并发检查该组里所有行的“有效位（Valid == 1）”与“标记位（Tag 比对）”，如果匹配成功即为命中，锁定具体是哪一行（Line/Way）。
3、字抽取（Word/Byte Extraction）：最后利用低 bbb 位（如 6 位）作为块内偏移量，从该行 64 字节数据中直接取出那 1 个目标字节返回给寄存器。

回收内存带来的性能影响以及常见的解决方式

回收有两种方式：
1、后代内存回收，唤醒kswapd线程，这种是异步回收不会阻塞进程。
2、直接内存回收，这种方式是同步回收，会阻塞进程造成长时间延迟，CPU利用率升高。

可回收内存类型为文件页和匿名页：
文件页回收：干净页直接回收，不影响性能。脏页先写回磁盘在释放。
匿名页回收：开启Swap机制，将不常访问匿名页换出磁盘，下次在访问再换入内存中。

解决方式
调整文件页和匿名页回收倾向,/proc/sys/vm/swappiness 这个值越大越积极Swap，繁殖倾向于回收文件页。
尽早触发kswapd内核线程异步回收内存，调整/proc/sys/vm/min_free_kbytes 内存最低限，越小性能好，越大内存好。
NUMA架构下，通过 /proc/sys/vm/zone_reclaim_mode选取模式控制

0 （默认值）：在回收本地内存之前，在其他 Node 寻找空闲内存。
1：只回收本地内存干净页。
2：只回收本地内存，在本地回收内存时，可以将文件页中的脏页写回硬盘，以回收内存。
4：只回收本地内存，在本地回收内存时，可以用 swap 方式回收内存。

如果看到系统还有一般内存，但是频繁触发直接内存回收看看这个值是不是不是0.

如果保证一个重要进程不被杀，调小/proc/[pid]/oom_score_adj 这个（-1000~1000），彻底免疫 OOM Killer直接设置为-1000。

4GB物理内存申请8GB内存

要说明是32位系统，64位系统，有没有开启swap分区，申请8GB会不会被使用。
32位操作系统，进程理论最大申请3GB大小虚拟内存，所以会失败。
64位中，进程理论申请128T所以申请没关系，但是要看有没有swap分区
没有swap分区，物理空间不够会被操作系统杀掉，有swap分区即使物理内u才能有4GB，程序依然能使用8GB。

LRU和升级（双链表/分代） LRU

传统 LRU 盲目遵从“新即是热”，单次全量扫描（冷数据）就会直接将长期高频的热点彻底挤出；而升级版 LRU 划分“冷区（Inactive）”与“热区（Active）”，新数据仅能先入冷区，只有在冷区内再次被有效命中，才允许晋升至热区，以此筑起“防污染防火墙”，确保核心热数据不被单次突发流量冲垮。
文件页和预读的都先到inactive，文件页只有重复读取才提升到active。匿名页根据有page结构体mapping最低位位置为1确认，只要读取就放在active最低位。

进程调度原则

1、CPU利用率：调度程序确保CPU是匆忙状态，提高CPU利用率
2、系统吞吐量：“单位时间内CPU完成进程数量，长作业进程占用较长时间，降低吞吐量，相反短作业进程提升吞吐量。
3、周转时间：进程运行+阻塞+等待时间之和
4、等待时间：进程处于就绪队列，时间越长越不好
5、响应时间：用户请求到系统第一次产生响应所花费的时间，比如键盘鼠标

常用调度算法：先来先服务，最短作业优先，（高响应比优先调度（优先权=（等待时间/要求服务时间）+1），一般不知道时间所以只是理论算法），时间片轮转，最高优先级调度算法

linux有一个多级反馈队列算法，就是有一个就绪队列链表，优先级从高到底，优先级越高的时间片越短，每次有新任务默认最高优先级放在就绪队列1。在低优先级运行时，如果有高优先级来了立马打断去运行高优先级的，缺点低优先级会饿死，假如高优先级一直有任务，解决方案：每隔一秒把所有任务放在最高优先级。

| 维度 | 信号量 (Semaphore) | 条件变量 (Condition Variable) |
| :--- | :--- | :--- |
| 核心底色 | 数量管家 (拥有 5 个空闲 Buffer) | 状态监视器 (复杂事件是否发生) |

#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>
#include <time.h>

typedef struct {
    pthread_mutex_t mutex;
    pthread_cond_t    write_cond;
    pthread_cond_t    read_cond;
    int                                readers;
    int                                writing;
    int                                waiting_readers;
}reader_pref_rwlock_t;

void reader_pref_init(reader_pref_rwlock_t *lock) {
    pthread_mutex_init(&lock->mutex, NULL);
    pthread_cond_init(&lock->write_cond, NULL);
    pthread_cond_init(&lock->read_cond, NULL);
    lock->readers = 0;
    lock->writing = 0;
    lock->waiting_readers = 0;
}

// 读者加锁
void reader_pref_rdlock(reader_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->waiting_readers++;
    while (lock->writing != 0) {
        pthread_cond_wait(&lock->read_cond, &lock->mutex);
    }
    lock->waiting_readers--;
    lock->readers++;
    pthread_mutex_unlock(&lock->mutex);
}

// 读者解锁
void reader_pref_rdunlock(reader_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->readers--;
    if (lock->readers == 0) {
        pthread_cond_signal(&lock->write_cond);
    }
    pthread_mutex_unlock(&lock->mutex);
}

// 写者加锁
void reader_pref_wrlock(reader_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    while(lock->readers != 0 || lock->writing != 0) {
        pthread_cond_wait(&lock->write_cond, &lock->mutex);
    }
    lock->writing = 1;
    pthread_mutex_unlock(&lock->mutex);
}   
// 写者解锁
void reader_pref_wrunlock(reader_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->writing = 0;
    if (lock->waiting_readers)
        pthread_cond_broadcast(&lock->read_cond);
    else
        pthread_cond_signal(&lock->write_cond);
    pthread_mutex_unlock(&lock->mutex);
}// =================== 验证实验 ===================

reader_pref_rwlock_t my_lock;
int shared_data = 100;

void print_time(const char *role, const char *action) {
    time_t now = time(NULL);
    struct tm *t = localtime(&now);
    printf("[%02d:%02d:%02d] %s: %s\n", t->tm_hour, t->tm_min, t->tm_sec, role, action);
}

// 读者 1：0 秒时进场，耗时读取 3 秒（0s ~ 3s）
void* reader_1(void* arg) {
    reader_pref_rdlock(&my_lock);
    print_time("读者 1", "进入临界区，开始耗时读取 3 秒...");
    sleep(3);
    print_time("读者 1", "读取完毕，退出临界区");
    reader_pref_rdunlock(&my_lock);
    return NULL;
}

// 写者：在第 1 秒到达，尝试加锁排队
void* writer_thread(void* arg) {
    sleep(1); // 确保读者 1 先拿到锁
    print_time("写者", "到达！准备获取写锁（开始排队等读者 1）");
    reader_pref_wrlock(&my_lock);

    print_time("写者", "终于抢到锁！正在修改数据（耗时 2 秒）...");
    shared_data += 50;
    sleep(2);
    print_time("写者", "修改完成，释放写锁");

    reader_pref_wrunlock(&my_lock);
    return NULL;
}

// 读者 2（晚期读者）：在第 2 秒到达（此时读者 1 还没走，写者已经在排队等待）
void* reader_2(void* arg) {
    sleep(2); // 第 2 秒到达
    print_time("读者 2 (晚期)", "到达！尝试获取读锁...");

    // 在读者优先下：读者 2 发现 writing == 0（写者还在排队没进场），
    // 读者 2 会直接插队无视排队的写者，进去跟读者 1 一起并发读取！
    reader_pref_rdlock(&my_lock);

    print_time("读者 2 (晚期)", "成功获取读锁！开始并发读取 2 秒...");
    sleep(2);
    print_time("读者 2 (晚期)", "读取完毕，退出临界区");

    reader_pref_rdunlock(&my_lock);
    return NULL;
}

int main() {
    reader_pref_init(&my_lock);

    pthread_t r1, w, r2;
    pthread_create(&r1, NULL, reader_1, NULL);
    pthread_create(&w, NULL, writer_thread, NULL);
    pthread_create(&r2, NULL, reader_2, NULL);

    pthread_join(r1, NULL);
    pthread_join(w, NULL);
    pthread_join(r2, NULL);

    return 0;
}

#include <time.h>

typedef struct {
    pthread_mutex_t mutex;
    pthread_cond_t  read_cond;       // 读者等待队列
    pthread_cond_t  write_cond;      // 写者等待队列

    int readers;                     // 当前正在临界区读取的读者数量
    int writers;                     // 当前正在写入的写者数量 (0 或 1)
    int waiting_writers;             // 正在排队等待的写者数量 (核心标志)
    int waiting_readers;
} writer_pref_rwlock_t;

// 1. 初始化锁
void my_writer_pref_init(writer_pref_rwlock_t *lock) {
    pthread_mutex_init(&lock->mutex, NULL);
    pthread_cond_init(&lock->write_cond, NULL);
    pthread_cond_init(&lock->read_cond, NULL);
    lock->readers = 0;
    lock->writers = 0;
    lock->waiting_writers = 0;
    lock->waiting_readers = 0;
}

// 2. 读者加锁（写者优先：有写者排队时读者必须等待）
void my_writer_pref_rdlock(writer_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->waiting_readers++;
    while(lock->writers != 0 || lock->waiting_writers != 0) {
        pthread_cond_wait(&lock->read_cond, &lock->mutex);
    }
    lock->waiting_readers--;
    lock->readers++;
    pthread_mutex_unlock(&lock->mutex);
}

// 3. 读者解锁
void my_writer_pref_rdunlock(writer_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->readers--;
     if (lock->readers == 0 && lock->waiting_writers > 0) {
        pthread_cond_signal(&lock->write_cond);
     }


    pthread_mutex_unlock(&lock->mutex);
}

// 4. 写者加锁（声明排队，拦截后续读者）
void my_writer_pref_wrlock(writer_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    lock->waiting_writers++;
    while(lock->readers != 0 || lock->writers != 0) {
        pthread_cond_wait(&lock->write_cond, &lock->mutex);
    }
    lock->waiting_writers--;
    // 标记在写
    lock->writers = 1;
    pthread_mutex_unlock(&lock->mutex);
}

// 5. 写者解锁（优先唤醒下一个排队的写者）
void my_writer_pref_wrunlock(writer_pref_rwlock_t *lock) {
    pthread_mutex_lock(&lock->mutex);
    
     lock->writers = 0;
     if (lock->waiting_writers > 0) {
        pthread_cond_signal(&lock->write_cond);
     }
     // 真的有人在等待读 唤醒所有读者
     else if (lock->waiting_readers){
        pthread_cond_broadcast(&lock->read_cond);
     }
    pthread_mutex_unlock(&lock->mutex);
}

// =================== 测试验证平台 ===================

writer_pref_rwlock_t rwlock;
int shared_resource = 100;

void print_stamp(const char *role, const char *msg) {
    time_t now = time(NULL);
    struct tm *t = localtime(&now);
    printf("[%02d:%02d:%02d] %s: %s\n", t->tm_hour, t->tm_min, t->tm_sec, role, msg);
}

// 读者 1：0 秒启动，占用锁 3 秒（0s ~ 3s）
void* thread_reader1(void* arg) {
    my_writer_pref_rdlock(&rwlock);
    print_stamp("读者 1", "进入临界区，开始耗时读取 3 秒...");
    sleep(3);
    print_stamp("读者 1", "读取完成，退出临界区");
    my_writer_pref_rdunlock(&rwlock);
    return NULL;
}

// 写者：1 秒后到达并尝试加锁（排队等待读者 1）
void* thread_writer(void* arg) {
    sleep(1);
    print_stamp("写者", "到达，准备获取写锁（声明排队，拦截后续读者）");
    my_writer_pref_wrlock(&rwlock);
    print_stamp("写者", "成功获取写锁！开始写入（耗时 2 秒）...");
    shared_resource += 50;
    sleep(2);
    print_stamp("写者", "写入完成，释放写锁");
    my_writer_pref_wrunlock(&rwlock);
    return NULL;
}

// 读者 2：2 秒后到达（此时写者已在排队）
// 若写者优先成立：读者 2 必须被拦截在门外，等写者写完才能进入
void* thread_reader2(void* arg) {
    sleep(2);
    print_stamp("读者 2 (晚期)", "到达，尝试获取读锁...");
    my_writer_pref_rdlock(&rwlock);
    print_stamp("读者 2 (晚期)", "成功获取读锁！开始读取...");
    sleep(1);
    print_stamp("读者 2 (晚期)", "读取完成，退出临界区");
    my_writer_pref_rdunlock(&rwlock);
    return NULL;
}

int main() {
    my_writer_pref_init(&rwlock);

    pthread_t r1, w, r2;
    pthread_create(&r1, NULL, thread_reader1, NULL);
    pthread_create(&w, NULL, thread_writer, NULL);
    pthread_create(&r2, NULL, thread_reader2, NULL);

    pthread_join(r1, NULL);
    pthread_join(w, NULL);
    pthread_join(r2, NULL);

    printf("所有测试线程正常退出，写者优先验证完毕。\n");
    return 0;
}

优先级相同；写者、读者互斥访问；只能一个写者访问临界区；可以有多个读者同时访问临界资源； 具体可看小林coding的5.3

寻道算法

LOOK扫描算法（SCAN优化版，以物理极限点计算）：
方案 A（先左后右）：从起点往左走到最左端点，然后掉头一路向右走到最右端点。
总距离=（起点-最小值）+（最大值-最小值）
方案 B（先右后左）：从起点往右走到最右端点，然后掉头一路向左走到最左端点。
总距离=（最大值-起点）+（最大值-最小值）
在确定方向后开始移动，带移动过程中如果有相反方向点插入，不理他回程处理，相同方向点就处理。
缺点就是假如有一个反方向1距离的，反而要等待很久，而两端的能快速响应。

C-LOOK扫描算法（C-SCAN优化版，C-SCAN必须回到物理极限点）
先只处理一个方向，然后快速复位到最远点继续扫描，有优点就是时间固定，每一个优先级差不多。

select 和 poll 并没有本质区别，它们内部都是使用「线性结构」来存储进程关注的 Socket 集合。在使用的时候，首先需要把关注的 Socket 集合通过 select/poll 系统调用从用户态拷贝到内核态，然后由内核检测事件，当有网络事件产生时，内核需要遍历进程关注 Socket 集合，找到对应的 Socket，并设置其状态为可读/可写，然后把整个 Socket 集合从内核态拷贝到用户态，用户态还要继续遍历整个 Socket 集合找到可读/可写的 Socket，然后对其处理。
