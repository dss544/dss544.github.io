Linux 内存管理：PAGE_OFFSET 与空间划分

在 Linux 内存管理中，系统通过 PAGE_OFFSET 宏来界定用户空间与内核空间的分界线。
线性映射区（Direct Mapping Area）的大小，主要取决于 PAGE_OFFSET 的设置以及物理内存的实际容量。为了保证内核能灵活分配非连续的物理内存，系统必须在内核空间顶部为 vmalloc 区域保留一定的空间（通常至少保留 240MB）。

以下是两种常见的内存划分方案对比（32位linux系统）：

1. 经典 3GB/1GB 划分

在这种模式下，虚拟地址空间（4GB）被划分为 3GB 用户空间 + 1GB 内核空间。

  • 分界线 (PAGE_OFFSET)：3GB (0xC0000000)
  • 线性映射区：[3GB, 3GB + 784MB]
    (注：1024MB 内核空间 - 240MB vmalloc 保留 = 784MB)

2. 均衡 2GB/2GB 划分

在这种模式下，虚拟地址空间被对半划分，为内核提供更充裕的地址空间。

  • 分界线 (PAGE_OFFSET)：2GB (0x80000000)
  • 线性映射区：[2GB, 2GB + 1808MB]
    (注：2048MB 内核空间 - 240MB vmalloc 保留 = 1808MB)

💡 核心总结
无论采用哪种比例划分，内核空间的公式始终成立：
内核虚拟地址空间 = 线性映射区 + vmalloc保留区（≥240MB）+ 其他保留区

3. 深入探究：vmalloc 区域的边界与隔离机制

在内核编程中，最常用的两个内存分配函数是 kmalloc 和 vmalloc。

对比维度 kmalloc vmalloc
底层依赖 依赖 Slab 分配器（建立在伙伴系统之上） 直接依赖 伙伴系统 (Buddy System)
虚拟地址 连续 连续
物理地址 绝对连续 大概率不连续（由离散的物理页拼凑）
分配区域 线性映射区（Direct Mapping 区） vmalloc 保留区（位于内核空间较高地址处）
性能与速度 极快 (O(1)O(1)O(1))（直接返回地址，无需修改页表） 极慢（需寻找空闲物理页、修改页表、建立映射、刷新 TLB）
分配大小 适合小内存（通常 < PAGE_SIZE，极限不超过伙伴系统分配最大空间(4MB)） 适合大内存(可达几十 MB 甚至上 GB) vmalloc区域最小240M
中断上下文 支持（带 GFP_ATOMIC 标志可用于中断，不阻塞） 严禁在中断中使用（分配过程可能引发睡眠/阻塞）
典型使用场景 1. 硬件 DMA 缓冲区（必须物理连续）
2. 频繁申请/释放的核心数据结构（如 inode、task_struct）		 1. 加载庞大的内核模块（驱动代码）
2. 分配超大的软件缓冲区（如巨大的网络/视频缓冲池）

vmalloc只有用户申请的时候才回去做映射，它的工作流程，分配虚拟连续地址空间（遍历虚拟地址链表）根据内存大小从伙伴大小申请物理页帧，把物理页帧映射到虚拟地址空间，所以他自带阻塞属性，在使用
vmalloc时不需要像kmalloc一样添加标志位，代码如下。

static void *__vmalloc_area_node(struct vm_struct *area, gfp_t gfp_mask,
				 pgprot_t prot, int node)
{
	const int order = 0;
	struct page **pages;
	unsigned int nr_pages, array_size, i;
	/* 强制加上__GFP_ZERO确保分配内存为0 */
	const gfp_t nested_gfp = (gfp_mask & GFP_RECLAIM_MASK) | __GFP_ZERO;
	const gfp_t alloc_mask = gfp_mask | __GFP_NOWARN;
	/* 计算要申请4kb数量 */
	nr_pages = get_vm_area_size(area) >> PAGE_SHIFT;
	array_size = (nr_pages * sizeof(struct page *));

	area->nr_pages = nr_pages;
	/* Please note that the recursion is strictly bounded. */
	/* 要分配数组太大 连续空间放不下 只能递归申请                */
	if (array_size > PAGE_SIZE) {
		pages = __vmalloc_node(array_size, 1, nested_gfp|__GFP_HIGHMEM,
				PAGE_KERNEL, node, area->caller);
		area->flags |= VM_VPAGES;
	} else {
		/* 数组比较小直接走kmalloc */
		pages = kmalloc_node(array_size, nested_gfp, node);
	}
	area->pages = pages;
	if (!area->pages) {
		/* 申请失败了 */
		remove_vm_area(area->addr);
		kfree(area);
		return NULL;
	}
	for (i = 0; i < area->nr_pages; i++) {
		struct page *page;

		if (node == NUMA_NO_NODE)
			/* 不指定节点分配 */
			page = alloc_page(alloc_mask);
		else
			/* 指定节点分配 */
			page = alloc_pages_node(node, alloc_mask, order);

		if (unlikely(!page)) {
			/* Successfully allocated i pages, free them in __vunmap() */
			area->nr_pages = i;
			goto fail;
		}
		/* 申请到的单页指针存进数组 */
		area->pages[i] = page;
		/* 如果允许睡眠就调用内核调度器  */
		if (gfp_mask & __GFP_WAIT)
			cond_resched();
	}
	/* 映射物理地址到虚拟地址 */
	if (map_vm_area(area, prot, pages))
		goto fail;
	return area->addr;

fail:
	warn_alloc_failed(gfp_mask, order,
			  "vmalloc: allocation failure, allocated %ld of %ld bytes\n",
			  (area->nr_pages*PAGE_SIZE), area->size);
	vfree(area->addr);
	return NULL;
}

通过对比kmalloc和vmalloc，观察虚拟地址和实际物理地址区别。

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/slab.h>    /* kmalloc */
#include <linux/vmalloc.h> /* vmalloc */
#include <linux/mm.h>      /* vmalloc_to_page */
static unsigned char *p;
static int __init hello_init(void)
{
    unsigned int phy_addr;
    unsigned int pfn;
    int i;
    p = vmalloc(5 * 1024 * 1024);
    if(!p)
    {
        printk("error vmalloc!\n");
    }
    else
    {
        for (i = 0; i < 100; ++i)
        {
            /* vmalloc虚拟地址和物理地址是非线性 不能用virt_to_phys */
            pfn = vmalloc_to_pfn(p);
            phy_addr = (pfn << 12) | ((unsigned int)p & 0xfff);
            printk("virt_addr: %x pfn: %x phy_addr: %x\n",
                                (unsigned int)p, pfn, phy_addr);
            p += 4096;
        }
    }
    return 0;
}
static void __exit hello_exit(void)
{
    vfree(p);
}
module_init(hello_init);
module_exit(hello_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Driver_Developer");
MODULE_DESCRIPTION("A simple module to check kmalloc and vmalloc addresses");

程序运行结果如下，可以看到虚拟地址连续，物理地址不一定连续。

使用kmalloc分配内存，注意不能超过伙伴系统最大分配内存大小（这里是4MB），或则加载驱动直接报错。代码如下：

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/slab.h>    /* kmalloc */
#include <linux/vmalloc.h> /* vmalloc */
#include <linux/mm.h>      /* vmalloc_to_page */
static unsigned char *p;
static int __init hello_init(void)
{
    unsigned int phy_addr;
    unsigned int pfn;
    int i;
    p = kmalloc(4 * 1024 * 1024, GFP_KERNEL);
    if(!p)
    {
        printk("error vmalloc!\n");
    }
    else
    {
        for (i = 0; i < 100; ++i)
        {
            // 物理地址
            phy_addr = virt_to_phys(p);
            // 物理页帧
            pfn = phy_addr >> 12;           
            printk("virt_addr: %x pfn: %x phy_addr: %x\n",
                                (unsigned int)p, pfn, phy_addr);
            p += 4096;
        }
    }
    return 0;
}

static void __exit hello_exit(void)
{
    vfree(p);
}
module_init(hello_init);
module_exit(hello_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Driver_Developer");
MODULE_DESCRIPTION("A simple module to check kmalloc and vmalloc addresses");

运行结果如下，虚拟地址物理地址均连续

1. 核心理论：ARM32 两级页表与 4KB 小页映射 (Page Mapping)

在 ARM32 架构中，采用 4KB 小页映射时，硬件 MMU 使用两级页表结构：

一级页表 (L1 / PGD)

  • 包含 4096 个表项
  • 负责覆盖 4GB 的虚拟地址空间

二级页表 (L2 / PTE)

  • 包含 256 个表项
  • 每个表项占 4 字节
    因此整个二级页表大小刚好是：
256 × 4B = 1KB

2. 精妙的标志位设计（位复用）

既然二级页表大小是 1KB，那么它在物理内存中的起始地址必定是：

1KB 对齐

这意味着：

二级页表基地址低 10 位天然全是 0

因此：

  • 一级页表项只需要高 22 位 [31:10]
    来指向二级页表的物理基址
  • 腾出来的低 10 位 [9:0]
    就可以用于存放属性标志位（Flags）
    其中：
  • 最低 [1:0] 两位
    表示当前表项的映射类型
    例如：
    | 位值 | 类型 | 说明 |
    |---|---|---|
    | 00 | Fault | 无效映射 / 缺页异常 |
    | 01 | Table | 指向二级页表 |
    | 10 | Section | 直接映射 1MB 段 |

在代码：

arch/arm/include/asm/pgtable-2level-hwdef.h

中，定义了一级页表和二级页表项的标志位。

一级页表 PMD 标志位

#define PMD_TYPE_MASK   (_AT(pmdval_t, 3) << 0)  // 掩码 0b11
#define PMD_TYPE_FAULT  (_AT(pmdval_t, 0) << 0)  // 00: 触发缺页异常 (Ignored/Fault)
#define PMD_TYPE_TABLE  (_AT(pmdval_t, 1) << 0)  // 01: 指向二级页表 (Page table)
#define PMD_TYPE_SECT   (_AT(pmdval_t, 2) << 0)  // 10: 段映射 (Section, 1MB 大页)

二级页表 PTE 标志位

#define PTE_TYPE_SMALL  (_AT(pteval_t, 2) << 0) // 10: 4KB 小页
#define PTE_EXT_XN      (_AT(pteval_t, 1) << 0) // XN: Execute Never（不可执行，防止缓冲区溢出攻击）
#define PTE_BUFFERABLE  (_AT(pteval_t, 1) << 2) // B: 允许 Write Buffer
#define PTE_CACHEABLE   (_AT(pteval_t, 1) << 3) // C: 允许 Cache 缓存

__get_free_pages() 和 kmalloc()只要看到标志位有ZONE_HIGHMEM就会立即拒绝，因为这两个都是要返回虚拟地址，而高端内存没有固定虚拟地址，所以会拒绝。只有alloc_pages() 才能分配高端内存，他返回的值一个指向page的页，要自己kmap映射才能使用。

vmalloc可能会睡眠，指定标志位也没有
kfree里面有锁，不会睡眠。vfree会判断在不在中断，如果在中断就放在队列，不在就释放。所以也可以在中断使用。

slab是由一个或多个连续物理内存页，被切割成许多大小完全一样的房间，每个房间放一类对象
linux内核在初始化的时候，就建立的一个大数组struct kmem_cache *kmalloc_caches[KMALLOC_SHIFT_HIGH + 1];

kmap是用于映射一个page到内核地址空间，高端低端内存都可以用，如果page是地段内存，就会返回该页的虚拟地址，如果位于高端内存，就建立永久映射，永久映射数量有限，所以要kunmap再不需要高端内存的时候。
